Logical operator - SQL Queries

1. Display employees who work in department 90 and whose salary is greater than 10,000 or whose job_id is 'AD_VP'.
SELECT *
FROM EMPLOYEES
WHERE (DEPARTMENT_ID = 90 AND SALARY > 10000)
   OR JOB_ID = 'AD_VP';

2. List employees who work in department 60 or 100 and have a salary greater than 8,000.
SELECT *
FROM EMPLOYEES
WHERE (DEPARTMENT_ID = 60 OR DEPARTMENT_ID = 100)
  AND SALARY > 8000;

3. Display employees whose job_id is 'IT_PROG' and whose salary is less than 7,000 or who belong to department 60.
SELECT *
FROM EMPLOYEES
WHERE (JOB_ID = 'IT_PROG' AND SALARY < 7000)
   OR DEPARTMENT_ID = 60;

4. Show employees whose job_id is 'SA_REP' or 'SA_MAN' and who work in department 80.
SELECT *
FROM EMPLOYEES
WHERE (JOB_ID = 'SA_REP' OR JOB_ID = 'SA_MAN')
  AND DEPARTMENT_ID = 80;

5. Display employees who work in department 30 and whose salary is greater than 3,000 or whose manager_id is 114.
SELECT *
FROM EMPLOYEES
WHERE (DEPARTMENT_ID = 30 AND SALARY > 3000)
   OR MANAGER_ID = 114;

6. List employees who work in department 20 or 50 and have a salary less than 6,000.
SELECT *
FROM EMPLOYEES
WHERE (DEPARTMENT_ID = 20 OR DEPARTMENT_ID = 50)
  AND SALARY < 6000;

7. Display employees whose salary is greater than 9,000 and who work in department 90 or 100.
SELECT *
FROM EMPLOYEES
WHERE SALARY > 9000
  AND (DEPARTMENT_ID = 90 OR DEPARTMENT_ID = 100);

8. Show employees whose job_id is 'ST_CLERK' and who work in department 50 or whose salary is less than 3,000.
SELECT *
FROM EMPLOYEES
WHERE (JOB_ID = 'ST_CLERK' AND DEPARTMENT_ID = 50)
   OR SALARY < 3000;

9. Display employees whose salary is greater than 5,000 or who work in department 80.
SELECT *
FROM EMPLOYEES
WHERE SALARY > 5000
   OR DEPARTMENT_ID = 80;

10. List employees who work in department 60 and whose job_id is 'IT_PROG' or whose salary is greater than 10,000.
SELECT *
FROM EMPLOYEES
WHERE (DEPARTMENT_ID = 60 AND JOB_ID = 'IT_PROG')
   OR SALARY > 10000;
